-- ==================== 8. YUKI AIM ====================
local function acharAlvoMaisProximoMaximo()
	local char = LocalPlayer.Character
	if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
	local minhaPos = char.HumanoidRootPart.Position
	local maisProximo = nil
	local distanciaMin = math.huge
	for _, obj in pairs(workspace:GetDescendants()) do
		if obj:IsA("Humanoid") 
		   and obj.Health > 0 
		   and obj.Parent 
		   and obj.Parent:FindFirstChild("HumanoidRootPart") then
			local model = obj.Parent
			if model ~= char and Players:GetPlayerFromCharacter(model) ~= LocalPlayer then
				local dist = (model.HumanoidRootPart.Position - minhaPos).Magnitude
				if dist < distanciaMin then
					distanciaMin = dist
					maisProximo = model
				end
			end
		end
	end
	return maisProximo
end

-- ==================== 9. AUTO AQATING LOGIC ====================
local lastAqatingDodge = 0
local aqatingLocked = {}

RunService.Heartbeat:Connect(function()
	if not Config.AutoAqating then return end
	local myChar = LocalPlayer.Character
	local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
	if not myRoot then return end

	local function checkEnemyAnim(model)
		if not model or model == myChar then return end
		local hum = model:FindFirstChildOfClass("Humanoid")
		if not hum or hum.Health <= 0 then return end
		local er = model:FindFirstChild("HumanoidRootPart") or model:FindFirstChild("Torso")
		if not er or (myRoot.Position - er.Position).Magnitude > 25 then return end

		if aqatingLocked[model] and (tick() - aqatingLocked[model]) < 1.2 then return end

		local animator = hum:FindFirstChildOfClass("Animator")
		if not animator then return end

		for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
			local pri = track.Priority
			if track.WeightCurrent > 0.1 and (pri == Enum.AnimationPriority.Action or pri == Enum.AnimationPriority.Action2 or pri == Enum.AnimationPriority.Action3) then
				aqatingLocked[model] = tick()
				if tick() - lastAqatingDodge >= 0.5 then
					pcall(function()
						local main = LocalPlayer.PlayerGui:FindFirstChild("Main")
						local controls = main and main:FindFirstChild("Controls")
						local moveset = controls and controls:FindFirstChild("Moveset")
						if moveset then
							for _, skill in ipairs(moveset:GetChildren()) do
								local nc = string.lower(skill.Name)
								if nc:find("adapt") or nc:find("wheel") or nc:find("mahoraga") then
									local btn = skill:FindFirstChild("Touch") or skill:FindFirstChild("ItemName") or skill
									if btn and btn:IsA("GuiObject") and firesignal then
										firesignal(btn.MouseButton1Down)
										task.delay(0.05, function() pcall(firesignal, btn.MouseButton1Up) end)
									end
								end
							end
						end
					end)
					lastAqatingDodge = tick()
				end
				break
			end
		end
	end

	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LocalPlayer and p.Character then checkEnemyAnim(p.Character) end
	end
	local chars = workspace:FindFirstChild("Characters")
	if chars then
		for _, obj in ipairs(chars:GetChildren()) do
			if obj:IsA("Model") then checkEnemyAnim(obj) end
		end
	end
end)

-- ==================== 11. CORE COMBAT & SNAP CAMLOCK ====================
local function instantHardSnap(targetHRP)
	if not targetHRP or not targetHRP.Parent then return end
	local char = LocalPlayer.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	local humanoid = char and char:FindFirstChildOfClass("Humanoid")
	
	if hrp then
		if humanoid then humanoid.AutoRotate = false end
		local flatTarget = Vector3.new(targetHRP.Position.X, hrp.Position.Y, targetHRP.Position.Z)
		hrp.CFrame = CFrame.lookAt(hrp.Position, flatTarget)
		
		task.delay(0.3, function()
			if humanoid and not Config.SmartCamlock then
				humanoid.AutoRotate = true
			end
		end)
	end
end

RunService.RenderStepped:Connect(function()
	if not Config.SmartCamlock or not smartLockedHRP or not smartLockedHRP.Parent then return end
	local char = LocalPlayer.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	local humanoid = char and char:FindFirstChildOfClass("Humanoid")
	
	if hrp then
		if humanoid then humanoid.AutoRotate = false end
		local targetPos = smartLockedHRP.Position
		local flatTarget = Vector3.new(targetPos.X, hrp.Position.Y, targetPos.Z)
		hrp.CFrame = CFrame.lookAt(hrp.Position, flatTarget)
	else
		if humanoid then humanoid.AutoRotate = true end
	end
end)

local function triggerSkill(keyword)
	pcall(function()
		local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
		local main = playerGui and playerGui:FindFirstChild("Main")
		local controls = main and main:FindFirstChild("Controls")
		local moveset = controls and controls:FindFirstChild("Moveset")
		
		local targetButton = nil
		if moveset then
			for _, child in ipairs(moveset:GetChildren()) do
				if child.Name:lower():find(keyword) then
					targetButton = child:FindFirstChild("ItemName") or child:FindFirstChild("Touch") or child
					break
				end
			end
		end

		if targetButton and targetButton:IsA("GuiObject") and firesignal then
			firesignal(targetButton.MouseButton1Down)
			task.wait(0.02)
			firesignal(targetButton.MouseButton1Up)
		end
	end)
end

local isBlockActive, isParryActive, isSupernovaActive, isCounterActive = false, false, false, false

local function monitorRemote(remote)
	if not remote:IsA("RemoteEvent") then return end
	if remote == activatedRE or remote == deactivatedRE then return end
	if remote.Name:lower():find("emote") or remote.Name:lower():find("chat") then return end

	track(remote.OnClientEvent:Connect(function(...)
		local char = LocalPlayer.Character
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		if not hrp then return end

		for _, arg in pairs({ ... }) do
			if typeof(arg) == "Instance" then
				local model = arg:IsA("Model") and arg or arg:FindFirstAncestorOfClass("Model")
				if model and model ~= char then
					local eHRP = model:FindFirstChild("HumanoidRootPart")
					if eHRP then
						local dist = (hrp.Position - eHRP.Position).Magnitude
						
						if dist <= 30 then
							smartLockedHRP = eHRP
						end
						
						if Config.AutoBlock and dist <= Config.BlockDist and not isBlockActive and activatedRE and deactivatedRE then
							isBlockActive = true
							instantHardSnap(eHRP)
							pcall(function() activatedRE:FireServer() end)
							task.delay(Config.BlockDelay, function()
								pcall(function() deactivatedRE:FireServer() end)
								isBlockActive = false
							end)
						end
						
						if Config.AutoParry and dist <= Config.ParryDist and not isParryActive then
							isParryActive = true
							triggerSkill("outburst")
							task.delay(Config.ParryDelay, function() isParryActive = false end)
						end
						
						if Config.Supernova and dist <= Config.SupernovaDist and not isSupernovaActive then
							isSupernovaActive = true
							triggerSkill("supernova")
							task.delay(Config.SupernovaDelay, function() isSupernovaActive = false end)
						end
						
						if Config.AutoCounter and dist <= Config.CounterDist and not isCounterActive then
							isCounterActive = true
							triggerSkill("manji")
							task.delay(Config.CounterDelay, function() isCounterActive = false end)
						end
					end
				end
			end
		end
	end))
end

for _, descendant in pairs(ReplicatedStorage:GetDescendants()) do
	monitorRemote(descendant)
end
track(ReplicatedStorage.DescendantAdded:Connect(monitorRemote))

local function createSlider(parent, order, titleText, minVal, maxVal, currentVal, decimals, callback)
	local container = Instance.new("Frame")
	container.LayoutOrder = order
	container.Size = UDim2.new(1, -4, 0, 50)
	container.BackgroundColor3 = theme.elementBg
	container.Parent = parent
	applyCorner(container, 8)

	local label = Instance.new("TextLabel")
	label.Size = UDim2.new(1, -12, 0, 22)
	label.Position = UDim2.new(0, 12, 0, 4)
	label.BackgroundTransparency = 1
	label.TextColor3 = theme.text
	label.Font = Enum.Font.GothamMedium
	label.TextSize = 11
	label.TextXAlignment = Enum.TextXAlignment.Left
	label.Parent = container

	local function updateLabel(val)
		if decimals == 0 then
			label.Text = titleText .. ": " .. math.floor(val)
		else
			label.Text = string.format("%s: %.2f s", titleText, val)
		end
	end
	updateLabel(currentVal)

	local bar = Instance.new("Frame")
	bar.Position = UDim2.new(0.06, 0, 0, 32)
	bar.Size = UDim2.new(0.88, 0, 0, 6)
	bar.BackgroundColor3 = theme.inactive
	bar.Parent = container
	applyCorner(bar, 3)

	local fill = Instance.new("Frame")
	fill.Size = UDim2.new((currentVal - minVal) / (maxVal - minVal), 0, 1, 0)
	fill.BackgroundColor3 = theme.accent
	fill.Parent = bar
	applyCorner(fill, 3)

	local sliding = false
	bar.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			sliding = true
		end
	end)

	track(UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			sliding = false
		end
	end))

	track(UserInputService.InputChanged:Connect(function(input)
		if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local posPercentage = math.clamp((input.Position.X - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
			fill.Size = UDim2.new(posPercentage, 0, 1, 0)
			local calculatedVal = minVal + (maxVal - minVal) * posPercentage
			if decimals == 0 then
				calculatedVal = math.floor(calculatedVal + 0.5)
			end
			updateLabel(calculatedVal)
			callback(calculatedVal)
		end
	end))
end
