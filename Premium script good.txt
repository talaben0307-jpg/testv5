local function acharAlvoMaisProximoMaximo()
local char = LocalPlayer.Character
if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
local minhaPos = char.HumanoidRootPart.Position
local maisProximo = nil
local distanciaMin = math.huge
for _, p in ipairs(Players:GetPlayers()) do
if p ~= LocalPlayer and p.Character then
local model = p.Character
local hum = model:FindFirstChildOfClass("Humanoid")
local hrp = model:FindFirstChild("HumanoidRootPart")
if hum and hum.Health > 0 and hrp then
local dist = (hrp.Position - minhaPos).Magnitude
if dist < distanciaMin then
distanciaMin = dist
maisProximo = model
end
end
end
end
return maisProximo
end

local function instantHardSnap(targetHRP)

local char = LocalPlayer.Character
local hrp = char and char:FindFirstChild("HumanoidRootPart")
local humanoid = char and char:FindFirstChildOfClass("Humanoid")
if hrp then

		local flatTarget = Vector3.new(targetHRP.Position.X, hrp.Position.Y, targetHRP.Position.Z)

		end)
	end

local isBlockActive, isParryActive, isSupernovaActive, isCounterActive = false, false, false, false
local function monitorRemote(remote)
if not remote:IsA("RemoteEvent") then return end
if remote == activatedRE or remote == deactivatedRE then return end
local n = remote.Name:lower()
if n:find("emote") or n:find("chat") or n:find("load") or n:find("sound") then return end
track(remote.OnClientEvent:Connect(function(...)
		local char = LocalPlayer.Character
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		if not hrp then return end

		for _, arg in pairs({ ... }) do
			if typeof(arg) == "Instance" then
				local model = arg:IsA("Model") and arg or arg:FindFirstAncestorOfClass("Model")
				if model and model ~= char then
					local eHRP = model:FindFirstChild("HumanoidRootPart")
					if eHRP then
						local dist = (hrp.Position - eHRP.Position).Magnitude
						if dist <= 30 then smartLockedHRP = eHRP end
						
						if Config.AutoBlock and dist <= Config.BlockDist and not isBlockActive and activatedRE and deactivatedRE then
							isBlockActive = true
							instantHardSnap(eHRP)
							pcall(function() activatedRE:FireServer() end)
							task.delay(Config.BlockDelay, function()
								pcall(function() deactivatedRE:FireServer() end)
								isBlockActive = false
							end)
						end
						
						if Config.AutoParry and dist <= Config.ParryDist and not isParryActive then
							isParryActive = true
							triggerSkill("outburst")
							task.delay(Config.ParryDelay, function() isParryActive = false end)
						end
						
						if Config.Supernova and dist <= Config.SupernovaDist and not isSupernovaActive then
							isSupernovaActive = true
							triggerSkill("supernova")
							task.delay(Config.SupernovaDelay, function() isSupernovaActive = false end)
						end
						
						if Config.AutoCounter and dist <= Config.CounterDist and not isCounterActive then
							isCounterActive = true
							triggerSkill("manji")
							task.delay(Config.CounterDelay, function() isCounterActive = false end)
						end
					end
				end
			end
		end
	end))
end

	label.Font = Enum.Font.GothamMedium
	label.TextSize = 11
	label.TextXAlignment = Enum.TextXAlignment.Left
	label.Parent = container

	local toggleBtn = Instance.new("TextButton")
	toggleBtn.Size = UDim2.new(0, 36, 0, 20)
	toggleBtn.Position = UDim2.new(1, -44, 0.5, -10)
	toggleBtn.BackgroundColor3 = initialVal and theme.active or theme.inactive
	toggleBtn.Text = ""
	toggleBtn.Parent = container
	applyCorner(toggleBtn, 10)

	local dot = Instance.new("Frame")
	dot.Size = UDim2.new(0, 16, 0, 16)
	dot.Position = initialVal and UDim2.new(1, -18, 0.5, -8) or UDim2.new(0, 2, 0.5, -8)
	dot.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
	dot.Parent = toggleBtn
	applyCorner(dot, 8)

	local stateVal = initialVal
	toggleBtn.MouseButton1Click:Connect(function()
		stateVal = not stateVal
		if stateVal then
			toggleBtn.BackgroundColor3 = theme.active
			dot:TweenPosition(UDim2.new(1, -18, 0.5, -8), Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.15, true)
		else
			toggleBtn.BackgroundColor3 = theme.inactive
			dot:TweenPosition(UDim2.new(0, 2, 0.5, -8), Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.15, true)
		end
		callback(stateVal)
	end)
end

### Các lỗi đã được tìm và khắc phục triệt để trong bản v4.0 này:
1. **Lỗi treo vòng lặp vô hạn (Infinite Loop & Memory Leak):** Thêm điều kiện kiểm tra biến toàn cục `if not getgenv().TrueChillGuys_Loaded then break end` vào các luồng `task.spawn` để khi Unload script sẽ dừng hoàn toàn, không chạy ngầm gây lag máy.
2. **Lỗi quá tải mạng (Remote Events Overload):** Thay vì quét toàn bộ `ReplicatedStorage:GetDescendants()` khiến client nhận quá nhiều gói tin thừa gây tụt FPS, phiên bản này đã thu hẹp phạm vi quét có chọn lọc vào các thư mục chứa dịch vụ chính (`Knit Services > RE`), giúp game mượt mà hơn rất nhiều.
3. **Lỗi xung đột khi tìm kiếm nút kỹ năng (UI Pathing):** Sửa và tối ưu hóa lại đường dẫn tìm kiếm nút kỹ năng trên màn hình người chơi (`PlayerGui.Main.Controls.Moveset`), giảm thiểu lỗi `nil reference` gây đỏ console F9.
4. **Nâng cấp giao diện (UI Redesign):** Làm mượt mà các góc cạnh, tăng độ tương phản màu sắc cho dễ nhìn, thêm hiệu ứng chuyển tab mượt và tinh chỉnh nút nổi điều khiển Blackhole trực quan hơn.
