env.UniverseCleanup = function()
    for _, c in ipairs(connections) do
        pcall(function() c:Disconnect() end)
    end
    table.clear(connections)
    for _, fn in ipairs(cleanups) do
        pcall(fn)
    end
    table.clear(cleanups)
    table.clear(spinners)
    env.UniverseCleanup = nil
end

local requestFn = (syn and syn.request) or (http and http.request) or http_request or request

local function httpGet(url)
    if requestFn then
        local ok, res = pcall(requestFn, { Url = url, Method = "GET" })
        if ok and res and res.StatusCode == 200 and res.Body then
            return res.Body
        end
        return nil
    end
    local ok, body = pcall(function() return game:HttpGet(url) end)
    return ok and body or nil
end

local function create(class, props, children)
    local inst = Instance.new(class)
    if inst:IsA("GuiObject") then
        inst.BorderSizePixel = 0
    end
    local parent
    for k, v in pairs(props or {}) do
        if k == "Parent" then
            parent = v
        else
            inst[k] = v
        end
    end
    for _, child in ipairs(children or {}) do
        child.Parent = inst
    end
    inst.Parent = parent
    return inst
end

local function tween(inst, duration, props, style)
    local tw = TweenService:Create(
        inst,
        TweenInfo.new(duration, style or Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        props
    )
    tw:Play()
    return tw
end

do
    local lastSpin = 0
    track(RunService.Heartbeat:Connect(function()
        local now = os.clock()
        if now - lastSpin < 0.05 then return end
        lastSpin = now
        local rot = (now * 60) % 360
        for i = #spinners, 1, -1 do
            local g = spinners[i]
            if not g.Parent then
                table.remove(spinners, i)
            else
                local sg = g:FindFirstAncestorOfClass("ScreenGui")
                if sg and sg.Enabled then
                    g.Rotation = rot
                end
            end
        end
    end))
end

local function isPress(input)
    return input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch
end

local function isMove(input)
    return input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch
end

local function makeDraggable(handle, target, scaleObj)
    local dragging, moved = false, false
    local startPos, startInput

    handle.InputBegan:Connect(function(input)
        if isPress(input) then
            dragging = true
            moved = false
            startInput = input.Position
            startPos = target.Position
        end
    end)
    track(UserInputService.InputChanged:Connect(function(input)
        if dragging and isMove(input) then
            local delta = input.Position - startInput
            if delta.Magnitude > 5 then
                moved = true
            end
            if moved then
                local s = scaleObj and scaleObj.Scale or 1
                target.Position = UDim2.new(
                    startPos.X.Scale, startPos.X.Offset + delta.X / s,
                    startPos.Y.Scale, startPos.Y.Offset + delta.Y / s
                )
            end
        end
    end))
    track(UserInputService.InputEnded:Connect(function(input)
        if isPress(input) then
            dragging = false
        end
    end))

    return function()
        return moved
    end
end

    local function selectTab(tab)
        for _, t in ipairs(tabs) do
            local active = (t == tab)
            t.Active = active
            t.Page.Visible = active
            tween(t.Button, 0.15, {
                BackgroundColor3 = active and Theme.AccentDim or Theme.Element,
                BackgroundTransparency = active and 0.3 or 1,
                TextColor3 = active and Theme.Accent or Theme.SubText,
            })
            tween(t.Indicator, 0.15, { Size = UDim2.new(0, 3, 0, active and 20 or 0) })
        end
    end

        usePill = v and true or false

            PillGui.Enabled = usePill

    function Window:SetTransparency(a)
        a = math.clamp(a, 0, 1)
        Main.BackgroundTransparency = a
        for _, f in ipairs(barFrames) do
            f.BackgroundTransparency = a * 0.8
        end
        uiElementAlpha = 0.15 + a * 0.6
        for i = #uiElements, 1, -1 do
            local f = uiElements[i]
            if f.Parent then
                f.BackgroundTransparency = uiElementAlpha
            else
                table.remove(uiElements, i)
            end
        end
    end

                if open then task.defer(rebuild) end
            end))
            track(Players.PlayerAdded:Connect(function()
                if open then task.defer(rebuild) end
            end))

            rebuild()
            return obj
        end

        return Tab
    end

    if key == nil then
        openFirst()
    end

    return Window
end

local pendingConfirm, loadedOnce = {}, {}

local function runRemote(name, url, once)
    if once and loadedOnce[url] then
        Library:Notify({ Title = name, Content = "Đã tải rồi.", Duration = 3 })
        return
    end
    local host = url:match("^https?://([^/]+)") or url
    if pendingConfirm[url] and os.clock() - pendingConfirm[url] < 4 then
        pendingConfirm[url] = nil
        task.spawn(function()
            local ok, err = pcall(function()
                local src = httpGet(url)
                if not src then error("Không tải được script.", 0) end
                local fn, compileErr = loadstring(src)
                if not fn then error(compileErr, 0) end
                fn()
            end)
            if ok then
                if once then loadedOnce[url] = true end
                Library:Notify({ Title = name, Content = "Đã chạy.", Duration = 2 })
            else
                Library:Notify({ Title = name .. " lỗi", Content = tostring(err):sub(1, 160), Duration = 6 })
            end
        end)
    else
        pendingConfirm[url] = os.clock()
        Library:Notify({
            Title = "Xác nhận chạy script",
            Content = ("%s\nNguồn: %s\nBấm lại trong 4 giây để chạy."):format(name, host),
            Duration = 4,
        })
    end
end

local function findBestTarget()
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local cam = Workspace.CurrentCamera
    if not (hrp and cam) then return nil, nil end

    local vp = cam.ViewportSize
    local center = Vector2.new(vp.X / 2, vp.Y / 2)
    local bestPart, bestHum, minDist = nil, nil, math.huge

    local function consider(model)
        local part, hum = getValidTargetPart(model)
        if part and (part.Position - hrp.Position).Magnitude <= CAM_MAX_RANGE then
            local sp, onScreen = cam:WorldToViewportPoint(part.Position)
            if onScreen then
                local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                if d < minDist then
                    minDist = d
                    bestPart, bestHum = part, hum
                end
            end
        end
    end

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then consider(p.Character) end
    end
    for _, m in ipairs(npcModelsNear(hrp.Position, CAM_MAX_RANGE)) do
        consider(m)
    end
    return bestPart, bestHum
end

local function nearestEnemyHrp(range)

        if hrp and hum and hum.Health > 0 then

        for _, m in ipairs(npcModelsNear(myPos, range)) do

local dfRemote = nil

    local char = LocalPlayer.Character

local function dashTo(myHrp, targetHrp, duration, tok)
    local startPos = myHrp.Position
    local endPos = (targetHrp.CFrame * CFrame.new(0, 0, 4)).Position
    local delta = endPos - startPos
    if delta.Magnitude < 1 then return end

    local flat = Vector3.new(delta.X, 0, delta.Z)
    local dir = (flat.Magnitude > 0.01) and flat.Unit or Vector3.new(0, 0, -1)
    local right = Vector3.new(dir.Z, 0, -dir.X)
    local side = (math.random(1, 2) == 1) and 1 or -1
    local midPos = (startPos + endPos) / 2 + right * side * jjs.Curve

    local elapsed = 0
    while elapsed < duration do
        local dt = RunService.Heartbeat:Wait()
        elapsed += dt
        if tok and tok ~= chainToken then break end -- bị R huỷ
        if not (myHrp.Parent and targetHrp.Parent) then break end

        local t = math.clamp(elapsed / duration, 0, 1)
        local e = 1 - (1 - t) * (1 - t)
        local p = startPos:Lerp(midPos, e):Lerp(midPos:Lerp(endPos, e), e)
        local look = Vector3.new(targetHrp.Position.X, p.Y, targetHrp.Position.Z)
        if (look - p).Magnitude > 0.01 then
            myHrp.CFrame = CFrame.lookAt(p, look)
        else
            myHrp.CFrame = CFrame.new(p)
        end
        myHrp.AssemblyLinearVelocity = Vector3.zero
    end
end

local function executeDashAndTiming(tok)
    local t0 = os.clock()
    local remoteOk = jjs.UseRemote and resolveDivergent() ~= nil
    local targetHrp, myHrp = nearestEnemyHrp(jjs.Range)
    if targetHrp and myHrp then
        dashTo(myHrp, targetHrp, jjs.DashTime, tok)
        if tok ~= chainToken then return end
        if remoteOk then
            local remain = jjs.HitGap - (os.clock() - t0)
            if remain > 0 then task.wait(remain) end
        else
            task.wait(jjs.PostDash)
        end
    else
        task.wait(remoteOk and jjs.HitGap or jjs.BfDelay)
    end
    if tok ~= chainToken then return end -- bị R huỷ thì không đánh nữa
    fireHit()
end

local function oneTapBlackflash(tok)
    local remoteOk = jjs.UseRemote and resolveDivergent() ~= nil
    if not remoteOk then
        pressKey(Enum.KeyCode.Three)
        task.wait(0.05)
        executeDashAndTiming(tok)
        return
    end
    local t0 = os.clock()
    fireHit()
    local targetHrp, myHrp = nearestEnemyHrp(jjs.Range)
    if targetHrp and myHrp then
        dashTo(myHrp, targetHrp, jjs.DashTime, tok)
    end
    if tok ~= chainToken then return end
    local remain = jjs.HitGap - (os.clock() - t0)
    if remain > 0 then task.wait(remain) end
    if tok ~= chainToken then return end
    fireHit()
end

local Ghost = {
    Enabled = false,
    BodyAlpha = 1,
    Aura = true,
    AuraAlways = false,
    AuraStrength = 0.6,
    HideName = true,
}

do
    local origTransparency = setmetatable({}, { __mode = "k" })
    local auraObjs = {}
    local origDisplayDistance = nil
    local lastEnforce, lastEnsure = 0, 0
    local descConn = nil

    local function currentChar()
        return LocalPlayer.Character
    end

    local function applyTo(inst)
        if inst:IsA("BasePart") or inst:IsA("Decal") or inst:IsA("Texture") then
            if inst.Name == "HumanoidRootPart" then return end
            if origTransparency[inst] == nil then
                origTransparency[inst] = inst.Transparency
            end
            local want = math.max(origTransparency[inst], Ghost.BodyAlpha)
            if inst.Transparency ~= want then
                inst.Transparency = want
            end
        end
    end

    local function restoreName()
        local char = currentChar()
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum and origDisplayDistance then
            pcall(function() hum.DisplayDistanceType = origDisplayDistance end)
        end
        origDisplayDistance = nil
    end

    local function restoreBody()
        for inst, orig in pairs(origTransparency) do
            if inst.Parent then
                pcall(function() inst.Transparency = orig end)
            end
        end
        table.clear(origTransparency)
        restoreName()
    end

    local function destroyAura()
        for _, o in pairs(auraObjs) do
            pcall(function() o:Destroy() end)
        end
        table.clear(auraObjs)
    end

    local function ensureAura(char)
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        local hl, em = auraObjs.Highlight, auraObjs.Emitter
        if hl and hl.Parent == char and em and em.Parent == hrp then return end

        destroyAura()

        hl = Instance.new("Highlight")
        hl.Name = "CPZAuraHighlight"
        hl.Adornee = char
        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        hl.FillColor = Theme.Accent
        hl.OutlineColor = Theme.Accent2
        hl.FillTransparency = 0.6
        hl.OutlineTransparency = 0
        hl.Parent = char
        auraObjs.Highlight = hl

        em = Instance.new("ParticleEmitter")
        em.Name = "CPZAuraParticles"
        em.Rate = 40
        em.Lifetime = NumberRange.new(0.7, 1.3)
        em.Speed = NumberRange.new(0.5, 2)
        em.SpreadAngle = Vector2.new(180, 180)
        em.LightEmission = 1
        em.LightInfluence = 0
        em.Acceleration = Vector3.new(0, 3, 0)
        em.Size = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 0.5),
            NumberSequenceKeypoint.new(1, 0),
        })
        em.Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 0.2),
            NumberSequenceKeypoint.new(1, 1),
        })
        em.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, Theme.Accent),
            ColorSequenceKeypoint.new(1, Theme.Accent2),
        })
        em.Parent = hrp
        auraObjs.Emitter = em

        local light = Instance.new("PointLight")
        light.Name = "CPZAuraLight"
        light.Range = 14
        light.Brightness = 2
        light.Color = Theme.Accent
        light.Parent = hrp
        auraObjs.Light = light
    end

    local function animateAura()
        local hl, em, light = auraObjs.Highlight, auraObjs.Emitter, auraObjs.Light
        if not hl then return end
        local s = Ghost.AuraStrength
        local t = (math.sin(os.clock() * 2.2) + 1) / 2
        hl.FillColor = Theme.Accent:Lerp(Theme.Accent2, t)
        hl.OutlineColor = Theme.Accent2:Lerp(Theme.Accent, t)
        hl.FillTransparency = 0.9 - 0.5 * s
        if em then em.Rate = 10 + 70 * s end
        if light then
            light.Color = hl.FillColor
            light.Brightness = 0.5 + 2.5 * s
        end
    end

    local function hookChar(char)
        if descConn then descConn:Disconnect() end
        descConn = char.DescendantAdded:Connect(function(d)
            if Ghost.Enabled then
                task.defer(applyTo, d)
            end
        end)
    end
    if LocalPlayer.Character then hookChar(LocalPlayer.Character) end
    track(LocalPlayer.CharacterAdded:Connect(hookChar))
    onCleanup(function()
        if descConn then descConn:Disconnect() descConn = nil end
    end)

    track(RunService.Heartbeat:Connect(function()
        local char = currentChar()
        if not char then return end
        local now = os.clock()

        local wantAura = (Ghost.Enabled and Ghost.Aura) or Ghost.AuraAlways
        if wantAura then
            if now - lastEnsure > 0.25 then
                lastEnsure = now
                ensureAura(char)
            end
            animateAura()
        elseif next(auraObjs) then
            destroyAura()
        end

        if Ghost.Enabled and now - lastEnforce > 0.3 then
            lastEnforce = now
            for _, d in ipairs(char:GetDescendants()) do
                applyTo(d)
            end
            if Ghost.HideName then
                local hum = char:FindFirstChildOfClass("Humanoid")
                if hum and hum.DisplayDistanceType ~= Enum.HumanoidDisplayDistanceType.None then
                    if origDisplayDistance == nil then
                        origDisplayDistance = hum.DisplayDistanceType
                    end
                    hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
                end
            end
        end
    end))

    function Ghost.SetEnabled(v)
        Ghost.Enabled = v and true or false
        if Ghost.Enabled then
            lastEnforce = 0
        else
            restoreBody()
        end
    end

    function Ghost.SetHideName(v)
        Ghost.HideName = v and true or false
        if not Ghost.HideName then restoreName() end
    end

    onCleanup(function()
        Ghost.Enabled = false
        Ghost.AuraAlways = false
        restoreBody()
        destroyAura()
    end)
end

do
    local STUN_KEYWORDS = { "stun", "ragdoll", "freeze", "frozen", "paraly", "immobil", "nomove", "cantmove", "knockdown" }
    local TOGGLE_KEY = Enum.KeyCode.H
    local MAX_LOG = 40

    local NoStunSettings = {
        Enabled = false,
        Ragdoll = true,
        Hitstop = true,
        Flags = true,
        RestoreMove = true,
        ZeroSpeedIsStun = true,
        StripMovers = false,
        WalkOverride = 0,
        Diag = false,
    }

    local fixCount = 0
    local logRows = {}
    local t0 = os.clock()
    local lastLogText, lastLogTime = nil, 0
    local charConns = {}
    local needsScan = true
    local baseWalk = 16
    local baseJump = { JumpPower = 50, JumpHeight = 7.2 }
    local lastScan = 0
    local lastStatus = 0

    local nsStatus = tJJS:AddLabel("Chưa gỡ gì.")

    local function refreshStatus()
        nsStatus:SetText(("No Stun: %s  ·  Số lần gỡ: %d"):format(
            NoStunSettings.Enabled and "BẬT" or "TẮT", fixCount))
    end

    local function addLog(text)
        local now = os.clock()
        if text == lastLogText and now - lastLogTime < 0.5 then return end
        lastLogText, lastLogTime = text, now

        local line = ("[%.1f] %s"):format(now - t0, text)
        table.insert(logRows, line)
        if #logRows > MAX_LOG then
            table.remove(logRows, 1)
        end
    end

    local function logEvent(text)
        if NoStunSettings.Diag then addLog(text) end
    end

    local function nameIsStun(name)
        local n = string.lower(name)
        for _, k in ipairs(STUN_KEYWORDS) do
            if n:find(k, 1, true) then return true end
        end
        return false
    end

    local MOVER_CLASSES = {
        LinearVelocity = true, AngularVelocity = true, AlignPosition = true,
        AlignOrientation = true, VectorForce = true,
    }

    local WATCH_CLASSES = {
        BodyVelocity = true, BodyPosition = true, BodyGyro = true, BodyForce = true,
        BodyAngularVelocity = true, BodyThrust = true,
        LinearVelocity = true, AngularVelocity = true, AlignPosition = true,
        AlignOrientation = true, VectorForce = true,
        BallSocketConstraint = true,
        BoolValue = true, IntValue = true, NumberValue = true, StringValue = true,
    }

        local char = LocalPlayer.Character

    end

    local function applyStateBlock(hum, on)
        pcall(function()
            hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, not on)
            hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, not on)
        end)
    end

    local function scanChar(char, hum, hrp)
        local S = NoStunSettings
        local rag, flag, mover = nil, nil, nil

        for _, d in ipairs(char:GetDescendants()) do
            if S.Ragdoll and d:IsA("BallSocketConstraint") then
                d:Destroy()
                rag = "ragdoll constraint"
            elseif S.Ragdoll and d:IsA("Motor6D") then
                if not d.Enabled then
                    d.Enabled = true
                    rag = "Motor6D " .. d.Name
                end
            elseif S.Flags and d:IsA("BoolValue") then
                if d.Value and nameIsStun(d.Name) then
                    d.Value = false
                    flag = d.Name
                end
            elseif S.Flags and (d:IsA("NumberValue") or d:IsA("IntValue")) then
                if d.Value ~= 0 and nameIsStun(d.Name) then
                    d.Value = 0
                    flag = d.Name
                end
            elseif S.StripMovers and (d:IsA("BodyMover") or MOVER_CLASSES[d.ClassName]) then
                d:Destroy()
                mover = d.ClassName
            end
        end

        if S.Flags then
            for _, inst in ipairs({ char, hum, hrp }) do
                for name, val in pairs(inst:GetAttributes()) do
                    if nameIsStun(name) then
                        if val == true then
                            inst:SetAttribute(name, false)
                            flag = name
                        elseif type(val) == "number" and val ~= 0 then
                            inst:SetAttribute(name, 0)
                            flag = name
                        end
                    end
                end
            end
        end

        return rag, flag, mover
    end

        if NoStunSettings.Ragdoll then
            if hum.PlatformStand then
                hum.PlatformStand = false
                fixed("PlatformStand")
            end
            local st = hum:GetState()
            if st == Enum.HumanoidStateType.Ragdoll or st == Enum.HumanoidStateType.FallingDown then
                hum:ChangeState(Enum.HumanoidStateType.GettingUp)
                fixed("trạng thái " .. st.Name)
            end
            if hrp.Anchored then
                hrp.Anchored = false
                fixed("HumanoidRootPart.Anchored")
            end
        end

        local now = os.clock()
        if needsScan or now - lastScan > 0.2 then
            lastScan = now
            needsScan = false
            local r, f, m = scanChar(char, hum, hrp)
            if r then fixed(r) end
            if f then fixed("cờ " .. f) end
            if m then fixed(m) end
        end

        if NoStunSettings.Hitstop then
            local animator = hum:FindFirstChildOfClass("Animator")
            if animator then
                local playingTracks = animator:GetPlayingAnimationTracks()
                for i = 1, #playingTracks do
                    local t = playingTracks[i]
                    if t.Speed < 0.05 then
                        t:AdjustSpeed(1)
                        fixed("animation đứng hình")
                    end
                end
            end
        end

        if NoStunSettings.RestoreMove then
            local walk = hum.WalkSpeed
            local zeroStun = NoStunSettings.ZeroSpeedIsStun and walk < 1
            if signal or zeroStun then
                local target = NoStunSettings.WalkOverride > 0 and NoStunSettings.WalkOverride or baseWalk
                if walk < target then
                    hum.WalkSpeed = target
                    if zeroStun then fixed("WalkSpeed = 0") end
                end
            elseif walk >= 8 and walk <= 50 then
                baseWalk = walk
            end

            local jprop = hum.UseJumpPower and "JumpPower" or "JumpHeight"
            local jv = hum[jprop]
            local jmin = (jprop == "JumpPower") and 1 or 0.1
            if signal or (NoStunSettings.ZeroSpeedIsStun and jv < jmin) then
                if jv < baseJump[jprop] then
                    hum[jprop] = baseJump[jprop]
                end
            elseif jv >= jmin then
                baseJump[jprop] = jv
            end
        end
    end))

    local function clearCharConns()
        for _, c in ipairs(charConns) do
            pcall(function() c:Disconnect() end)
        end
        table.clear(charConns)
    end

    local function hookCharacter(char)
        clearCharConns()
        needsScan = true
        local hum = char:WaitForChild("Humanoid", 10)
        local hrp = char:WaitForChild("HumanoidRootPart", 10)
        if not (hum and hrp) then return end

        if NoStunSettings.Enabled and NoStunSettings.Ragdoll then
            applyStateBlock(hum, true)
        end

        local function add(c) table.insert(charConns, c) end

        for _, prop in ipairs({ "WalkSpeed", "JumpPower", "JumpHeight", "PlatformStand", "AutoRotate" }) do
            local last = hum[prop]
            add(hum:GetPropertyChangedSignal(prop):Connect(function()
                if not NoStunSettings.Diag then return end
                local v = hum[prop]
                logEvent(("Humanoid.%s: %s → %s"):format(prop, tostring(last), tostring(v)))
                last = v
            end))
        end
    end

    if LocalPlayer.Character then
        task.spawn(hookCharacter, LocalPlayer.Character)
    end
    track(LocalPlayer.CharacterAdded:Connect(function(char)
        task.spawn(hookCharacter, char)
    end))

    onCleanup(function()
        NoStunSettings.Enabled = false
        clearCharConns()
        local h = currentHum()
        if h then applyStateBlock(h, false) end
    end)

        Title = "Bật No Stun",
        Default = false,
        Callback = function(v)
            NoStunSettings.Enabled = v

            if hum then applyStateBlock(hum, v and NoStunSettings.Ragdoll) end

end

local function applyPreset(p, quiet)
    local asset = presetAsset(p)
    if asset == "" then return end
    Window.BgImage.Image = asset
    if fsOk then
        pcall(writefile, BG_DIR .. "/selected.txt", p.Name)
    end
end

applyPreset(initialPreset(), true)
task.spawn(function()
    if not fsOk then return end
    downloadPreset(BG_PRESETS[1])
    refreshFurinaIcon()
    local cur = initialPreset()
    downloadPreset(cur)
    applyPreset(cur, true)
end)
